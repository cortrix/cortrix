# F06 Document Parser — Python bridge dependencies.
# Installed into the parser venv (real docling/paddleocr are D3.5 — not required
# for D3 standalone, where the C++ wrappers are tested against mock bridges).

# Docling structured parser (MIT, IBM): DocLayNet layout + TableFormer tables.
# Pinned: docling's bundled OCR engine (rapidocr) default changed across versions;
# 2.107.0 + onnxruntime below is the combo verified to OCR scanned/image PDFs.
docling==2.107.0

# rapidocr (docling's OCR) inference engine. WITHOUT onnxruntime, rapidocr 3.x
# falls back to its torch engine, which lacks the PP-OCRv6 models docling requests
# → "ValueError: Unsupported configuration: torch.PP-OCRv6.det.small" and every
# PDF/scanned/image parse fails (DEFECT#6). onnxruntime is rapidocr's default
# backend and ships working PP-OCR models.
onnxruntime>=1.20.0

# PaddleOCR fallback for scanned / image documents (F06 S3).
paddleocr>=2.8
# Verified window on Linux x86_64 CPU (2026-08-12, PP-OCRv6 models):
#   3.0.x  -> model attribute incompatibility (strides not Int32Attribute)
#   3.1/3.2 -> inference correct (native stdout noise handled by the bridges)
#   3.3.1  -> hard crash: ConvertPirAttribute2RuntimeAttribute unsupported
#             pir::ArrayAttribute<DoubleAttribute> (onednn instruction path)
paddlepaddle>=3.1,<3.3
